cmake_minimum_required(VERSION 3.16)
project(cpp_cron LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_cron tests/test_cron.cpp)
target_include_directories(test_cron PRIVATE include)
if(MSVC)
  target_compile_options(test_cron PRIVATE /W4)
else()
  target_compile_options(test_cron PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_cron COMMAND test_cron)
